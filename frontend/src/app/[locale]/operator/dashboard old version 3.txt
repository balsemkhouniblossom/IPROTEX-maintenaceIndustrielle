"use client";
import { useEffect, useMemo, useState } from "react";
import { useParams, useRouter } from "next/navigation";
import DashboardLayout from "@/components/DashboardLayout";
import ProtectedRoute from "@/components/auth/ProtectedRoute";
import { useAuth } from "@/contexts/AuthContext";
import { useTranslations } from "next-intl";
import {
    BellAlertIcon,
    CheckCircleIcon,
    ClockIcon,
    ExclamationTriangleIcon,
    ArrowRightIcon,
    CalendarDaysIcon,
    ClipboardDocumentListIcon,
    WrenchScrewdriverIcon,
    SparklesIcon,
    DocumentTextIcon,
    CogIcon
} from '@heroicons/react/24/outline';
import { apiService } from "@/services/api";

interface OperatorStats {
    waitingValidation: number;
    assigned: number;
    inProgress: number;
    overdue: number;
    completed: number;
}

interface CalendarWidgetCounts {
    today: number;
    thisWeek: number;
    nextWeek: number;
    nextMonth: number;
    overdue: number;
    waitingValidation: number;
}

interface WorkOrderItem {
    _id: string;
    ot_id?: string;
    status?: string;
    type_maintenance?: string;
    date_created?: string;
    date_start?: string;
    machine_id?: string | { _id?: string; machine_id?: string };
    description?: string;
    technician_id?: string | { _id?: string; nom_complet?: string };
}

interface InterventionReportItem {
    _id: string;
    report_id?: string;
    ot_id?: string | { _id?: string };
    description_action?: string;
    validation_responsable?: string;
    date_debut?: string;
    date_fin?: string;
}

interface CalendarEventItem {
    id: string;
    title: string;
    status: string;
    dueDate: string;
    priority?: string;
    machine?: { code?: string };
    frequency?: { label?: string };
}

function extractId(value: unknown): string {
    if (!value) return "";
    if (typeof value === "string") return value;
    if (typeof value === "object") {
        const objectValue = value as { _id?: string; id?: string };
        return objectValue._id || objectValue.id || "";
    }
    return "";
}

function extractMachineLabel(value: WorkOrderItem["machine_id"]): string {
    if (!value) return "-";
    if (typeof value === "string") return value;
    return value.machine_id || value._id || "-";
}

function normalizeApiItems<T>(payload: unknown): T[] {
    if (Array.isArray(payload)) {
        return payload as T[];
    }

    if (payload && typeof payload === "object") {
        const maybeItems = (payload as { items?: unknown }).items;
        if (Array.isArray(maybeItems)) {
            return maybeItems as T[];
        }
    }

    return [];
}

export default function OperatorDashboard() {
    const tOperator = useTranslations("dashboard.operator");
    const tCommon = useTranslations("common");
    const { user, isLoading: authLoading } = useAuth();
    const router = useRouter();
    const params = useParams();
    const locale = Array.isArray(params?.locale)
        ? params.locale[0]
        : params?.locale || "en";

    const [workOrders, setWorkOrders] = useState<WorkOrderItem[]>([]);
    const [reports, setReports] = useState<InterventionReportItem[]>([]);
    const [calendarEvents, setCalendarEvents] = useState<CalendarEventItem[]>([]);
    const [stats, setStats] = useState<OperatorStats>({
        waitingValidation: 0,
        assigned: 0,
        inProgress: 0,
        overdue: 0,
        completed: 0
    });
    const [calendarCounts, setCalendarCounts] = useState<CalendarWidgetCounts>({
        today: 0,
        thisWeek: 0,
        nextWeek: 0,
        nextMonth: 0,
        overdue: 0,
        waitingValidation: 0,
    });
    const [loading, setLoading] = useState(true);
    const today = useMemo(() => new Date(), []);

    const todaysDateLabel = useMemo(
        () => new Intl.DateTimeFormat(locale, {
            weekday: "long",
            day: "2-digit",
            month: "long",
            year: "numeric",
        }).format(today),
        [locale, today],
    );

    const shiftLabel = useMemo(() => {
        const hour = today.getHours();
        if (hour >= 6 && hour < 14) {
            return tOperator("shift.morning");
        }
        if (hour >= 14 && hour < 22) {
            return tOperator("shift.afternoon");
        }
        return tOperator("shift.night");
    }, [today, tOperator]);

    const activeWorkOrders = useMemo(
        () => workOrders.filter((order) => order.technician_id && extractId(order.technician_id) === user?._id),
        [user?._id, workOrders],
    );

    const derivedMachineLabel = useMemo(() => {
        const firstActive = activeWorkOrders.find((order) => order.status !== "completed");
        return firstActive ? extractMachineLabel(firstActive.machine_id) : tOperator("dashboard.emptyValue");
    }, [activeWorkOrders, tOperator]);

    const nextMaintenance = useMemo(() => {
        const upcoming = [...calendarEvents]
            .filter((event) => event.status !== "completed")
            .sort((left, right) => new Date(left.dueDate).getTime() - new Date(right.dueDate).getTime())[0];

        if (!upcoming) {
            return null;
        }

        return upcoming;
    }, [calendarEvents]);

    const recentReports = useMemo(() => {
        return reports
            .filter((report) => {
                const workOrder = workOrders.find((order) => order._id === extractId(report.ot_id));
                return !workOrder || !user?._id || extractId(workOrder.technician_id) === user._id;
            })
            .slice()
            .sort((left, right) => new Date(right.date_fin || right.date_debut || 0).getTime() - new Date(left.date_fin || left.date_debut || 0).getTime())
            .slice(0, 4);
    }, [reports, workOrders, user?._id]);

    const scheduleBuckets = useMemo(() => {
        const now = new Date();
        const endOfDay = new Date(now);
        endOfDay.setHours(23, 59, 59, 999);
        const tomorrow = new Date(now);
        tomorrow.setDate(tomorrow.getDate() + 1);
        const endOfWeek = new Date(now);
        endOfWeek.setDate(endOfWeek.getDate() + 7);

        const upcoming = calendarEvents.filter((event) => event.status !== "completed");

        return {
            today: upcoming.filter((event) => {
                const due = new Date(event.dueDate);
                return due <= endOfDay;
            }),
            upcoming: upcoming.filter((event) => new Date(event.dueDate) > endOfDay && new Date(event.dueDate) <= endOfWeek),
            overdue: upcoming.filter((event) => event.status === "overdue" || new Date(event.dueDate).getTime() < now.getTime()),
            completed: calendarEvents.filter((event) => event.status === "completed"),
            tomorrow,
        };
    }, [calendarEvents]);

    const notifications = useMemo(() => {
        return [
            { label: tOperator("dashboardNotifications.dueToday"), count: calendarCounts.today, tone: "orange" as const },
            { label: tOperator("dashboardNotifications.overdue"), count: calendarCounts.overdue, tone: "red" as const },
            { label: tOperator("dashboardNotifications.awaitingValidation"), count: calendarCounts.waitingValidation, tone: "violet" as const },
        ];
    }, [calendarCounts, tOperator]);

    const statsCards = [
        { label: tOperator("stats.assignedToYou"), value: stats.assigned, icon: ClipboardDocumentListIcon, tone: "blue" },
        { label: tOperator("stats.inProgress"), value: stats.inProgress, icon: WrenchScrewdriverIcon, tone: "emerald" },
        { label: tOperator("stats.waitingValidation"), value: stats.waitingValidation, icon: ClockIcon, tone: "violet" },
        { label: tOperator("stats.overdue"), value: stats.overdue, icon: ExclamationTriangleIcon, tone: "red" },
        { label: tOperator("stats.completedToday"), value: stats.completed, icon: CheckCircleIcon, tone: "green" },
    ];

    useEffect(() => {
        if (authLoading) {
            return;
        }
        if (!user?._id) {
            return;
        }

        let cancelled = false;

        const loadOperatorStats = async () => {
            try {
                const [workOrdersResponse, widgetResponse, reportsResponse, calendarResponse] = await Promise.all([
                    apiService.getWorkOrders(),
                    apiService.getCalendarWidget(),
                    apiService.getInterventionReports(),
                    apiService.getCalendarEvents({ view: "week", date: new Date().toISOString().slice(0, 10) }),
                ]);

                if (cancelled) {
                    return;
                }

                const normalizedWorkOrders = normalizeApiItems<WorkOrderItem>(workOrdersResponse.data);
                const reportItems = normalizeApiItems<InterventionReportItem>(reportsResponse.data);
                const calendarPayload = (calendarResponse.data as { items?: unknown } | undefined)?.items ?? calendarResponse.data;
                const calendarItems = normalizeApiItems<CalendarEventItem>(calendarPayload);

                setWorkOrders(normalizedWorkOrders);
                setReports(reportItems);
                setCalendarEvents(calendarItems);

                const scopedWorkOrders = normalizedWorkOrders.filter((order) => extractId(order.technician_id) === user._id);
                const waitingValidation = scopedWorkOrders.filter((wo) => wo.status === "waiting_validation").length;
                const assigned = scopedWorkOrders.filter((wo) => wo.status === "assigned").length;
                const inProgress = scopedWorkOrders.filter((wo) => wo.status === "in_progress").length;
                const completed = scopedWorkOrders.filter((wo) => wo.status === "completed").length;

                const counts = (widgetResponse.data as { counts?: CalendarWidgetCounts }).counts;
                setCalendarCounts({
                    today: Number(counts?.today || 0),
                    thisWeek: Number(counts?.thisWeek || 0),
                    nextWeek: Number(counts?.nextWeek || 0),
                    nextMonth: Number(counts?.nextMonth || 0),
                    overdue: Number(counts?.overdue || 0),
                    waitingValidation: Number(counts?.waitingValidation || 0),
                });

                setStats({
                    waitingValidation,
                    assigned,
                    inProgress,
                    overdue: Number(counts?.overdue || 0),
                    completed,
                });
            } catch (error) {
                console.error("Error loading operator stats", error);
            } finally {
                if (!cancelled) {
                    setLoading(false);
                }
            }
        };

        void loadOperatorStats();

        return () => {
            cancelled = true;
        };
    }, [user?._id, authLoading]);


    if (loading) {
        return (
            <div className="operator-dashboard-theme min-h-screen flex items-center justify-center bg-slate-50">
                <div className="rounded-3xl border border-slate-200 bg-white px-8 py-10 shadow-sm">
                    <div className="animate-spin rounded-full h-14 w-14 border-b-2 border-slate-900 mx-auto"></div>
                    <div className="mt-4 text-sm font-medium text-slate-600">{tCommon("loading")}</div>
                </div>
            </div>
        );
    }

    return (
        <ProtectedRoute requiredRole="operator">
            <DashboardLayout title={tOperator("title")}>
                <div className="operator-dashboard-theme bento-grid">
                    <section className="col-span-full rounded-4xl border border-slate-200 bg-linear-to-br from-white via-slate-50 to-amber-50 p-6 shadow-sm">
                        <div className="flex flex-col gap-6 lg:flex-row lg:items-end lg:justify-between">
                            <div className="max-w-3xl">
                                <div className="inline-flex items-center gap-2 rounded-full border border-slate-200 bg-white px-3 py-1 text-xs font-semibold uppercase tracking-[0.2em] text-slate-600">
                                    <SparklesIcon className="h-4 w-4" />
                                    {tOperator("dashboard.operatorConsole")}
                                </div>
                                <h1 className="mt-4 text-3xl font-bold tracking-tight text-slate-900 md:text-4xl">
                                    {tOperator("welcome", { name: user?.nom_complet || tCommon("defaultUserName") })}
                                </h1>
                                <p className="mt-3 max-w-2xl text-sm leading-6 text-slate-600">
                                    {tOperator("dashboard.intro")}
                                </p>
                            </div>
                            <div className="grid gap-3 sm:grid-cols-3">
                                <div className="rounded-2xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
                                    <div className="text-[11px] font-semibold uppercase tracking-[0.18em] text-slate-500">{tOperator("dashboard.todayDate")}</div>
                                    <div className="mt-2 text-sm font-semibold text-slate-900">{todaysDateLabel}</div>
                                </div>
                                <div className="rounded-2xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
                                    <div className="text-[11px] font-semibold uppercase tracking-[0.18em] text-slate-500">{tOperator("dashboard.currentShift")}</div>
                                    <div className="mt-2 text-sm font-semibold text-slate-900">{shiftLabel}</div>
                                </div>
                                <div className="rounded-2xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
                                    <div className="text-[11px] font-semibold uppercase tracking-[0.18em] text-slate-500">{tOperator("dashboard.machineAssigned")}</div>
                                    <div className="mt-2 text-sm font-semibold text-slate-900">{derivedMachineLabel}</div>
                                </div>
                            </div>
                        </div>

                        <div className="mt-6 grid gap-3 md:grid-cols-3 xl:grid-cols-6">
                            {notifications.map((item) => (
                                <div
                                    key={item.label}
                                    className={`rounded-2xl border p-4 shadow-sm ${item.tone === "red" ? "border-red-200 bg-red-50 text-red-900" : item.tone === "violet" ? "border-violet-200 bg-violet-50 text-violet-900" : "border-amber-200 bg-amber-50 text-amber-900"}`}
                                >
                                    <div className="text-xs font-semibold uppercase tracking-wide opacity-80">{item.label}</div>
                                    <div className="mt-2 text-2xl font-bold">{item.count}</div>
                                </div>
                            ))}
                        </div>
                    </section>

                    <section className="col-span-full">
                        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-5">
                            {statsCards.map((card) => {
                                const Icon = card.icon;
                                return (
                                    <div key={card.label} className="panel">
                                        <div className="flex items-start justify-between gap-3">
                                            <div>
                                                <div className="text-sm text-slate-500">{card.label}</div>
                                                <div className="mt-2 text-3xl font-bold text-slate-900">{card.value}</div>
                                            </div>
                                            <div className={`rounded-2xl p-3 ${card.tone === "red" ? "bg-red-50 text-red-600" : card.tone === "green" ? "bg-emerald-50 text-emerald-600" : card.tone === "violet" ? "bg-violet-50 text-violet-600" : card.tone === "emerald" ? "bg-emerald-50 text-emerald-600" : "bg-blue-50 text-blue-600"}`}>
                                                <Icon className="h-6 w-6" />
                                            </div>
                                        </div>
                                    </div>
                                );
                            })}
                        </div>
                    </section>

                    <section className="col-span-full panel">
                        <div className="card-title mb-4">{tOperator("dashboard.primaryActions")}</div>
                        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-3 auto-rows-fr items-stretch">
                                <button
                                onClick={() => router.push(`/${locale}/operator/preventive`)}
                                className="group w-full min-w-0 h-full rounded-3xl border border-blue-200 bg-blue-50 p-6 text-left transition hover:-translate-y-1 hover:shadow-xl"
                            >
                                <div className="flex h-full items-start justify-between gap-4">
                                    <div>
                                        <div className="text-lg font-bold text-blue-900">{tOperator("preventiveMaintenance")}</div>
                                        <p className="mt-2 text-sm leading-6 text-blue-800/80 line-clamp-4 break-words">{tOperator("dashboard.preventiveDescription")}</p>
                                    </div>
                                    <ArrowRightIcon className="h-6 w-6 text-blue-700 transition group-hover:translate-x-1" />
                                </div>
                            </button>
                                <button
                                onClick={() => router.push(`/${locale}/operator/corrective`)}
                                className="group w-full min-w-0 h-full rounded-3xl border border-emerald-200 bg-emerald-50 p-6 text-left transition hover:-translate-y-1 hover:shadow-xl"
                            >
                                <div className="flex h-full items-start justify-between gap-4">
                                    <div>
                                        <div className="text-lg font-bold text-emerald-900">{tOperator("correctiveMaintenance")}</div>
                                        <p className="mt-2 text-sm leading-6 text-emerald-800/80 line-clamp-4 break-words">{tOperator("dashboard.correctiveDescription")}</p>
                                    </div>
                                    <ArrowRightIcon className="h-6 w-6 text-emerald-700 transition group-hover:translate-x-1" />
                                </div>
                            </button>
                                <button
                                onClick={() => router.push(`/${locale}/operator/smart-maintenance-calendar`)}
                                className="group w-full min-w-0 h-full rounded-3xl border border-slate-200 bg-slate-900 p-6 text-left text-white transition hover:-translate-y-1 hover:shadow-xl"
                            >
                                <div className="flex h-full items-start justify-between gap-4">
                                    <div>
                                        <div className="text-lg font-bold">{tOperator("quickActions.smartCalendar")}</div>
                                        <p className="mt-2 text-sm leading-6 text-white/75 line-clamp-4 break-words">{tOperator("dashboard.calendarDescription")}</p>
                                    </div>
                                    <ArrowRightIcon className="h-6 w-6 text-white transition group-hover:translate-x-1" />
                                </div>
                            </button>
                        </div>
                    </section>

                    <section className="col-span-full grid grid-cols-1 gap-6 xl:grid-cols-2 xl:items-start">
                        <div className="panel min-w-0">
                            <div className="mb-4 flex items-center justify-between gap-3">
                                <div>
                                    <div className="card-title">{tOperator("dashboard.todaySchedule")}</div>
                                    <p className="mt-1 text-sm text-slate-500">{tOperator("dashboard.scheduleDescription")}</p>
                                </div>
                                <CalendarDaysIcon className="h-6 w-6 text-slate-400" />
                            </div>
                            <div className="grid gap-4 grid-cols-2 xl:grid-cols-4 auto-rows-fr">
                                {[
                                    { label: tOperator("dashboard.scheduleToday"), value: scheduleBuckets.today.length, tone: "orange" },
                                    { label: tOperator("dashboard.scheduleUpcoming"), value: scheduleBuckets.upcoming.length, tone: "blue" },
                                    { label: tOperator("dashboard.scheduleOverdue"), value: scheduleBuckets.overdue.length, tone: "red" },
                                    { label: tOperator("dashboard.scheduleCompleted"), value: scheduleBuckets.completed.length, tone: "green" },
                                ].map((item) => (
                                    <div key={item.label} className={`min-h-[92px] rounded-2xl border p-4 ${item.tone === "red" ? "border-red-200 bg-red-50 text-red-900" : item.tone === "green" ? "border-emerald-200 bg-emerald-50 text-emerald-900" : item.tone === "orange" ? "border-amber-200 bg-amber-50 text-amber-900" : "border-blue-200 bg-blue-50 text-blue-900"}`}>
                                        <div className="text-xs font-semibold uppercase tracking-wide opacity-80">{item.label}</div>
                                        <div className="mt-2 text-2xl font-bold">{item.value}</div>
                                    </div>
                                ))}
                            </div>
                            <div className="mt-5 space-y-3">
                                {scheduleBuckets.today.slice(0, 3).map((event) => (
                                    <div key={event.id} className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
                                        <div className="flex items-start justify-between gap-4">
                                            <div>
                                                <div className="font-semibold text-slate-900">{event.title}</div>
                                                <div className="mt-1 text-sm text-slate-600">
                                                    {event.machine?.code || tOperator("dashboard.emptyValue")} · {event.frequency?.label || tOperator("dashboard.emptyValue")}
                                                </div>
                                            </div>
                                            <div className="rounded-full border border-slate-200 px-3 py-1 text-xs font-semibold uppercase tracking-wide text-slate-600">
                                                {event.priority || tOperator("dashboard.noPriority")}
                                            </div>
                                        </div>
                                        <div className="mt-3 text-sm text-slate-500">
                                            {tOperator("dashboard.dueAt")} {new Intl.DateTimeFormat(locale, { hour: "2-digit", minute: "2-digit", day: "2-digit", month: "short" }).format(new Date(event.dueDate))}
                                        </div>
                                    </div>
                                ))}
                                {scheduleBuckets.today.length === 0 ? (
                                    <div className="rounded-2xl border border-dashed border-slate-300 bg-slate-50 p-5 text-sm text-slate-500">
                                        {tOperator("dashboard.noScheduledItems")}
                                    </div>
                                ) : null}
                            </div>
                        </div>

                        <div className="min-w-0 space-y-6">
                            <div className="panel">
                                <div className="card-title mb-4">{tOperator("dashboard.nextScheduledMaintenance")}</div>
                                {nextMaintenance ? (
                                    <div className="space-y-3">
                                        <div className="text-lg font-semibold text-slate-900">{nextMaintenance.title}</div>
                                        <div className="text-sm text-slate-600">{nextMaintenance.machine?.code || tOperator("dashboard.emptyValue")}</div>
                                        <div className="rounded-2xl border border-slate-200 bg-slate-50 p-4 text-sm text-slate-700">
                                            {tOperator("dashboard.dueAt")} {new Intl.DateTimeFormat(locale, { weekday: "short", day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" }).format(new Date(nextMaintenance.dueDate))}
                                        </div>
                                    </div>
                                ) : (
                                    <div className="text-sm text-slate-500">{tOperator("dashboard.noScheduledItems")}</div>
                                )}
                            </div>

                            <div className="panel">
                                <div className="mb-4 flex items-center justify-between gap-3">
                                    <div className="card-title">{tOperator("dashboard.currentNotifications")}</div>
                                    <BellAlertIcon className="h-6 w-6 text-amber-500" />
                                </div>
                                <div className="space-y-3">
                                    {notifications.map((item) => (
                                        <div key={item.label} className="flex items-center justify-between rounded-2xl border border-slate-200 bg-white px-4 py-3 text-sm shadow-sm">
                                            <span className="font-medium text-slate-700">{item.label}</span>
                                            <span className="rounded-full bg-slate-900 px-3 py-1 text-xs font-bold text-white">{item.count}</span>
                                        </div>
                                    ))}
                                </div>
                            </div>
                        </div>
                    </section>

                    <section className="col-span-full panel">
                        <div className="mb-4 flex items-center justify-between gap-3">
                            <div>
                                <div className="card-title">{tOperator("dashboard.recentReports")}</div>
                                <p className="mt-1 text-sm text-slate-500">{tOperator("dashboard.recentReportsDescription")}</p>
                            </div>
                            <ClipboardDocumentListIcon className="h-6 w-6 text-slate-400" />
                        </div>
                        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-4">
                            {recentReports.length === 0 ? (
                                <div className="text-sm text-slate-500">{tOperator("dashboard.noRecentReports")}</div>
                            ) : recentReports.map((report) => {
                                const workOrder = workOrders.find((order) => order._id === extractId(report.ot_id));
                                const machineId = workOrder ? extractMachineLabel(workOrder.machine_id) : tOperator("dashboard.emptyValue");
                                const maintenanceType = workOrder?.type_maintenance || tOperator("dashboard.emptyValue");
                                return (
                                    <button
                                        key={report._id}
                                        onClick={() => router.push(`/${locale}/operator/my-reports`)}
                                        className="rounded-[1.4rem] border border-slate-200 bg-white p-4 text-left shadow-sm transition hover:-translate-y-1 hover:shadow-lg"
                                    >
                                        <div className="flex items-start justify-between gap-3">
                                            <div>
                                                <div className="font-semibold text-slate-900">{report.report_id || tOperator("dashboard.emptyValue")}</div>
                                                <div className="mt-1 text-sm text-slate-500">{machineId}</div>
                                            </div>
                                            <div className="rounded-full border border-slate-200 px-3 py-1 text-[11px] font-semibold uppercase tracking-wide text-slate-600">
                                                {maintenanceType}
                                            </div>
                                        </div>
                                        <div className="mt-3 text-sm text-slate-700 line-clamp-3">
                                            {report.description_action || tOperator("dashboard.emptyValue")}
                                        </div>
                                    </button>
                                );
                            })}
                        </div>
                        <button
                            onClick={() => router.push(`/${locale}/operator/my-reports`)}
                            className="mt-5 inline-flex items-center gap-2 rounded-2xl bg-slate-900 px-4 py-3 text-sm font-semibold text-white transition hover:bg-slate-800"
                        >
                            <DocumentTextIcon className="h-4 w-4" />
                            {tOperator("viewAllReports")}
                        </button>
                    </section>

                    <section className="col-span-full panel">
                        <div className="mb-4 flex items-center justify-between gap-3">
                            <div>
                                <div className="card-title">{tOperator("machineManuals")}</div>
                                <p className="mt-1 text-sm text-slate-500">{tOperator("manualsIntro")}</p>
                            </div>
                            <CogIcon className="h-6 w-6 text-slate-400" />
